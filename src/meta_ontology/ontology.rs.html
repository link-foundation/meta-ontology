11/// A single concept in the meta-ontology.
12///
13/// Cycles between concepts are allowed and expected — `thing`, `concept`,
14/// and `link` all reference each other in seed data.
15#[derive(Debug, Clone, Default, PartialEq, Eq, Serialize, Deserialize, JsonSchema)]
16pub struct Concept {
17    /// Stable opaque identity. Names and labels may change without changing it.
18    pub id: ConceptId,
19    /// Canonical lowercase identifier, e.g. `thing`.
20    pub name: String,
21    /// Human‑readable label (English by default), e.g. `Thing`.
22    pub label: String,
23    /// Origin tag: `lino`, `nsm`, `schema`, `owl`, …
24    pub origin: String,
25    /// NSM category (substantive, quantifier, …) or empty.
26    pub category: String,
27    /// Allolexes (alternate surface forms in NSM theory).
28    pub allolexes: Vec<String>,
29    /// Explicit lookup aliases. Unlike NSM allolexes, these must be globally
30    /// unambiguous.
31    pub aliases: Vec<String>,
32    /// All definitions, in insertion order.
33    pub definitions: Vec<Definition>,
34    /// Cross‑ontology equivalences.
35    pub mappings: Vec<Mapping>,
36    /// Per‑language exponents loaded from `data/primes/exponents/<lang>.lino`.
37    pub exponents: BTreeMap<String, String>,
38    /// Evidence describing where this entity came from.
39    pub provenance: Provenance,
40    /// Ownership, lifecycle, classification, and tags.
41    pub governance: Governance,
42}
43
44/// One natural‑language definition of a concept.
45///
46/// A definition is a flat list of words (concept names) — the loader
47/// preserves them in document order.
48#[derive(Debug, Clone, PartialEq, Eq, Serialize, Deserialize, JsonSchema)]
49pub struct Definition {
50    /// Ordered concept words used by the definition.
51    pub words: Vec<String>,
52}
53
54/// A cross‑ontology mapping — `thing ~ schema_Thing` and friends.
55#[derive(Debug, Clone, PartialEq, Eq, Serialize, Deserialize, JsonSchema)]
56pub struct Mapping {
57    /// External name like `schema_Thing`, `wikidata_Q35120`.
58    pub external: String,
59    /// Relation kind: `~` (equivalent), `<` (narrower), `>` (broader).
60    pub kind: String,
61}

63/// In‑memory representation of the meta‑ontology.
64///
65/// Built by [`crate::loader`] from `.lino` files.
66#[derive(Debug, Clone)]
67pub struct Ontology {
68    /// Version of the public catalog/interchange contract.
69    pub(crate) schema_version: u32,
70    /// All concepts, keyed by canonical name. `BTreeMap` keeps `names()`
71    /// alphabetical without an extra sort.
72    pub(crate) concepts: BTreeMap<String, Concept>,
73    /// Explicit, typed edges between stable concept identities.
74    pub(crate) relationships: Vec<Relationship>,
75    /// Words that never need a concept — proper nouns, identifiers, etc.
76    pub(crate) allowlist: BTreeSet<String>,
77    /// Languages declared in any exponent file.
78    pub(crate) languages: BTreeSet<String>,
79    /// Normalized links-network representation supplied by `meta-language`.
80    pub(crate) network: LinkNetwork,
81}
82
83impl Default for Ontology {
84    fn default() -> Self {
85        Self {
86            schema_version: CURRENT_SCHEMA_VERSION,
87            concepts: BTreeMap::new(),
88            relationships: Vec::new(),
89            allowlist: BTreeSet::new(),
90            languages: BTreeSet::new(),
91            network: LinkNetwork::new(),
92        }
93    }
94}

96impl Ontology {
97    pub(crate) fn concept_mut_or_insert(
98        &mut self,
99        name: &str,
100        source_uri: &str,
101        source_line: usize,
102    ) -> &mut Concept {
103        self.concepts
104            .entry(name.to_string())
105            .or_insert_with(|| Concept {
106                id: ConceptId::from_name(name),
107                name: name.to_string(),
108                provenance: Provenance::for_source(source_uri, source_line),
109                ..Concept::default()
110            })
111    }
112
113    /// Iterate every concept name, alphabetically.
114    pub fn names(&self) -> impl Iterator<Item = &str> {
115        self.concepts.keys().map(String::as_str)
116    }
117
118    /// Look up a concept by name (case sensitive — names are canonical
119    /// lowercase identifiers).
120    #[must_use]
121    pub fn find(&self, name: &str) -> Option<&Concept> {
122        self.concepts.get(name)
123    }
124
125    /// Number of concepts in the ontology.
126    #[must_use]
127    pub fn len(&self) -> usize {
128        self.concepts.len()
129    }
130
131    /// Whether the ontology has no concepts.
132    #[must_use]
133    pub fn is_empty(&self) -> bool {
134        self.concepts.is_empty()
135    }
136
137    /// Iterate the names of concepts referenced by `name`'s definitions
138    /// (out‑neighbours in the definition graph).
139    ///
140    /// Words that are not themselves concept names are skipped. Returns
141    /// an empty iterator if `name` is unknown.
142    pub fn neighbors<'a>(&'a self, name: &'a str) -> impl Iterator<Item = &'a Concept> + 'a {
143        self.concepts
144            .get(name)
145            .into_iter()
146            .flat_map(|c| c.definitions.iter())
147            .flat_map(|def| def.words.iter())
148            .filter_map(move |word| self.concepts.get(word.as_str()))
149    }
150
151    /// Check whether a lower‑cased word is allow‑listed (proper noun,
152    /// identifier, version string, etc.).
153    #[must_use]
154    pub fn is_allowed(&self, word: &str) -> bool {
155        self.allowlist.contains(word)
156    }
157
158    /// Whether a lower‑cased word is either a known concept (or its
159    /// allolex) or on the allow‑list.
160    #[must_use]
161    pub fn covers(&self, word: &str) -> bool {
162        if self.concepts.contains_key(word) || self.is_allowed(word) {
163            return true;
164        }
165        // Match allolexes
166        self.concepts
167            .values()
168            .any(|c| c.allolexes.iter().any(|a| a == word))
169    }
170
171    /// Iterate all languages with at least one declared exponent.
172    pub fn languages(&self) -> impl Iterator<Item = &str> {
173        self.languages.iter().map(String::as_str)
174    }
175}

214    #[test]
215    fn cycle_exists_thing_concept_thing() {
216        let o = load_from_dir(data_dir()).expect("load");
217        // Walk neighbours of `thing` — `concept` should be reachable, and
218        // walking from `concept` should reach `thing` again, completing a
219        // cycle. We only need to assert one step each direction.
220        let from_thing: Vec<&str> = o.neighbors("thing").map(|c| c.name.as_str()).collect();
221        assert!(
222            from_thing.contains(&"concept"),
223            "thing -> concept must exist, got {from_thing:?}"
224        );
225        let from_concept: Vec<&str> = o.neighbors("concept").map(|c| c.name.as_str()).collect();
226        assert!(
227            from_concept.contains(&"thing"),
228            "concept -> thing must exist (cycle), got {from_concept:?}"
229        );
230    }
