3use std::collections::{BTreeMap, BTreeSet};

5/// A single concept in the meta-ontology.
6///
7/// Cycles between concepts are allowed and expected — `thing`, `concept`,
8/// and `link` all reference each other in seed data.
9#[derive(Debug, Clone, Default, PartialEq, Eq)]
10pub struct Concept {
11    /// Canonical lowercase identifier, e.g. `thing`.
12    pub name: String,
13    /// Human‑readable label (English by default), e.g. `Thing`.
14    pub label: String,
15    /// Origin tag: `lino`, `nsm`, `schema`, `owl`, …
16    pub origin: String,
17    /// NSM category (substantive, quantifier, …) or empty.
18    pub category: String,
19    /// Allolexes (alternate surface forms in NSM theory).
20    pub allolexes: Vec<String>,
21    /// All definitions, in insertion order.
22    pub definitions: Vec<Definition>,
23    /// Cross‑ontology equivalences.
24    pub mappings: Vec<Mapping>,
25    /// Per‑language exponents loaded from `data/primes/exponents/<lang>.lino`.
26    pub exponents: BTreeMap<String, String>,
27}
28
29/// One natural‑language definition of a concept.
30///
31/// A definition is a flat list of words (concept names) — the loader
32/// preserves them in document order.
33#[derive(Debug, Clone, PartialEq, Eq)]
34pub struct Definition {
35    pub words: Vec<String>,
36}
37
38/// A cross‑ontology mapping — `thing ~ schema_Thing` and friends.
39#[derive(Debug, Clone, PartialEq, Eq)]
40pub struct Mapping {
41    /// External name like `schema_Thing`, `wikidata_Q35120`.
42    pub external: String,
43    /// Relation kind: `~` (equivalent), `<` (narrower), `>` (broader).
44    pub kind: String,
45}
46
47/// In‑memory representation of the meta‑ontology.
48///
49/// Built by [`crate::loader`] from `.lino` files.
50#[derive(Debug, Clone, Default)]
51pub struct Ontology {
52    /// All concepts, keyed by canonical name. `BTreeMap` keeps `names()`
53    /// alphabetical without an extra sort.
54    pub(crate) concepts: BTreeMap<String, Concept>,
55    /// Words that never need a concept — proper nouns, identifiers, etc.
56    pub(crate) allowlist: BTreeSet<String>,
57    /// Languages declared in any exponent file.
58    pub(crate) languages: BTreeSet<String>,
59}
60
61impl Ontology {
62    /// Iterate every concept name, alphabetically.
63    pub fn names(&self) -> impl Iterator<Item = &str> {
64        self.concepts.keys().map(String::as_str)
65    }
66
67    /// Look up a concept by name (case sensitive — names are canonical
68    /// lowercase identifiers).
69    #[must_use]
70    pub fn find(&self, name: &str) -> Option<&Concept> {
71        self.concepts.get(name)
72    }
73
74    /// Number of concepts in the ontology.
75    #[must_use]
76    pub fn len(&self) -> usize {
77        self.concepts.len()
78    }
79
80    /// Whether the ontology has no concepts.
81    #[must_use]
82    pub fn is_empty(&self) -> bool {
83        self.concepts.is_empty()
84    }
85
86    /// Iterate the names of concepts referenced by `name`'s definitions
87    /// (out‑neighbours in the definition graph).
88    ///
89    /// Words that are not themselves concept names are skipped. Returns
90    /// an empty iterator if `name` is unknown.
91    pub fn neighbors<'a>(&'a self, name: &'a str) -> impl Iterator<Item = &'a Concept> + 'a {
92        self.concepts
93            .get(name)
94            .into_iter()
95            .flat_map(|c| c.definitions.iter())
96            .flat_map(|def| def.words.iter())
97            .filter_map(move |word| self.concepts.get(word.as_str()))
98    }
99
100    /// Check whether a lower‑cased word is allow‑listed (proper noun,
101    /// identifier, version string, etc.).
102    #[must_use]
103    pub fn is_allowed(&self, word: &str) -> bool {
104        self.allowlist.contains(word)
105    }
106
107    /// Whether a lower‑cased word is either a known concept (or its
108    /// allolex) or on the allow‑list.
109    #[must_use]
110    pub fn covers(&self, word: &str) -> bool {
111        if self.concepts.contains_key(word) || self.is_allowed(word) {
112            return true;
113        }
114        // Match allolexes
115        self.concepts
116            .values()
117            .any(|c| c.allolexes.iter().any(|a| a == word))
118    }
119
120    /// Iterate all languages with at least one declared exponent.
121    pub fn languages(&self) -> impl Iterator<Item = &str> {
122        self.languages.iter().map(String::as_str)
123    }
124}

126#[cfg(test)]
127mod tests {
128    use crate::loader::load_from_dir;
129
130    fn data_dir() -> std::path::PathBuf {
131        let manifest_dir = env!("CARGO_MANIFEST_DIR");
132        std::path::PathBuf::from(manifest_dir).join("data")
133    }
134
135    #[test]
136    fn loads_seed_data() {
137        let o = load_from_dir(data_dir()).expect("load");
138        assert!(o.find("thing").is_some(), "thing must be in seed data");
139        assert!(o.find("concept").is_some(), "concept must be in seed data");
140        assert!(o.find("link").is_some(), "link must be in seed data");
141    }
142
143    #[test]
144    fn cycle_exists_thing_concept_thing() {
145        let o = load_from_dir(data_dir()).expect("load");
146        // Walk neighbours of `thing` — `concept` should be reachable, and
147        // walking from `concept` should reach `thing` again, completing a
148        // cycle. We only need to assert one step each direction.
149        let from_thing: Vec<&str> = o.neighbors("thing").map(|c| c.name.as_str()).collect();
150        assert!(
151            from_thing.contains(&"concept"),
152            "thing -> concept must exist, got {from_thing:?}"
153        );
154        let from_concept: Vec<&str> = o.neighbors("concept").map(|c| c.name.as_str()).collect();
155        assert!(
156            from_concept.contains(&"thing"),
157            "concept -> thing must exist (cycle), got {from_concept:?}"
158        );
159    }
160
161    #[test]
162    fn nsm_primes_present() {
163        let o = load_from_dir(data_dir()).expect("load");
164        for prime in [
165            "i", "you", "someone", "people", "body", "kind", "part", "this", "other", "one", "two",
166            "good", "bad", "big", "small", "know", "think", "want", "feel", "see", "hear", "say",
167            "words", "true", "do", "happen", "move", "live", "die", "now", "here", "not", "if",
168            "very", "more", "like",
169        ] {
170            assert!(o.find(prime).is_some(), "missing NSM prime: {prime}");
171        }
172    }
173
174    #[test]
175    fn multiple_definitions_preserved() {
176        let o = load_from_dir(data_dir()).expect("load");
177        let thing = o.find("thing").expect("thing");
178        assert!(
179            thing.definitions.len() >= 2,
180            "thing must have multiple definitions, got {}",
181            thing.definitions.len()
182        );
183    }
184
185    #[test]
186    fn english_exponents_loaded() {
187        let o = load_from_dir(data_dir()).expect("load");
188        let lang_codes: Vec<&str> = o.languages().collect();
189        assert!(
190            lang_codes.contains(&"en"),
191            "english must be present, got {lang_codes:?}"
192        );
193    }
194
195    #[test]
196    fn allowlist_rejects_non_listed_word() {
197        let o = load_from_dir(data_dir()).expect("load");
198        assert!(o.is_allowed("github"));
199        assert!(!o.is_allowed("totallyimaginarywordxyz"));
200    }
